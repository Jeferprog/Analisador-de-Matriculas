// Chaves de propriedades do script para persistência segura
const PROP_API_KEY = 'GEMINI_API_KEY';
const PROP_CONFIG = 'CRESOL_CONFIG_JSON';

// Custo: modelo e preços do Gemini (USD por 1M de tokens) + câmbio de fallback
const GEMINI_MODEL_NAME = 'gemini-2.5-flash';
const PRECO_INPUT_USD_1M = 0.30;   // US$ / 1M tokens de entrada
const PRECO_OUTPUT_USD_1M = 2.50;  // US$ / 1M tokens de saída (inclui tokens de raciocínio)
const CAMBIO_FALLBACK = 5.40;      // usado se a cotação ao vivo falhar

// Controle de acesso por perfil (baseado no e-mail logado)
const PROP_ADMINS = 'CRESOL_ADMINS_JSON';
const ADMIN_INICIAL = 'jeferson.deimling@cresolsicoper.com.br';

/**
 * E-mail do usuário logado (domínio Cresol), em minúsculas. Vazio se não for possível identificar.
 */
function getUsuarioAtual() {
  try {
    const email = Session.getActiveUser().getEmail();
    return email ? email.trim().toLowerCase() : '';
  } catch (e) {
    return '';
  }
}

/**
 * Lista de e-mails administradores. Garante o admin inicial sempre presente.
 */
function getAdmins() {
  const props = PropertiesService.getScriptProperties();
  let lista = [];
  const raw = props.getProperty(PROP_ADMINS);
  if (raw) {
    try { lista = JSON.parse(raw); } catch (e) { lista = []; }
  }
  if (!Array.isArray(lista)) lista = [];
  const inicial = ADMIN_INICIAL.toLowerCase();
  const jaTem = lista.some(function(e) { return String(e).trim().toLowerCase() === inicial; });
  if (!jaTem) {
    lista.unshift(ADMIN_INICIAL);
    props.setProperty(PROP_ADMINS, JSON.stringify(lista));
  }
  return lista;
}

/**
 * Indica se um e-mail é administrador.
 */
function isAdmin(email) {
  if (!email) return false;
  const e = String(email).trim().toLowerCase();
  return getAdmins().some(function(a) { return String(a).trim().toLowerCase() === e; });
}

/**
 * Sessão do usuário para a interface: e-mail, se é admin e (para admins) a lista de admins.
 */
function getSessao() {
  const email = getUsuarioAtual();
  const admin = isAdmin(email);
  return {
    email: email,
    isAdmin: admin,
    admins: admin ? getAdmins() : []
  };
}

/**
 * Barreira de segurança: lança erro se o chamador não for administrador.
 * Usada nas funções sensíveis (Importação IA e Configurações).
 */
function exigirAdmin() {
  const email = getUsuarioAtual();
  if (!isAdmin(email)) {
    const detalhe = email
      ? ' (' + email + ')'
      : '. Não foi possível identificar seu e-mail — verifique se está logado com a conta Cresol e se o app está publicado com acesso restrito ao domínio.';
    throw new Error('Acesso restrito: apenas administradores podem executar esta ação' + detalhe);
  }
  return email;
}

/**
 * Cadastra um novo administrador (somente admin).
 */
function adicionarAdmin(email) {
  exigirAdmin();
  const e = String(email || '').trim().toLowerCase();
  if (!e || e.indexOf('@') === -1) throw new Error('E-mail inválido.');
  const lista = getAdmins();
  if (!lista.some(function(x) { return String(x).trim().toLowerCase() === e; })) {
    lista.push(e);
    PropertiesService.getScriptProperties().setProperty(PROP_ADMINS, JSON.stringify(lista));
  }
  return { success: true, admins: lista, message: 'Administrador ' + e + ' cadastrado com sucesso.' };
}

/**
 * Remove um administrador (somente admin). O admin principal não pode ser removido.
 */
function removerAdmin(email) {
  exigirAdmin();
  const e = String(email || '').trim().toLowerCase();
  if (e === ADMIN_INICIAL.toLowerCase()) {
    throw new Error('O administrador principal não pode ser removido.');
  }
  const lista = getAdmins().filter(function(x) { return String(x).trim().toLowerCase() !== e; });
  PropertiesService.getScriptProperties().setProperty(PROP_ADMINS, JSON.stringify(lista));
  return { success: true, admins: lista, message: 'Administrador ' + e + ' removido.' };
}

/**
 * Retorna a lista de chaves de API do Gemini cadastradas (uma por linha ou separadas
 * por vírgula/ponto-e-vírgula). Permite rotação automática quando uma chave atinge o limite.
 */
function getGeminiApiKeys() {
  const raw = PropertiesService.getScriptProperties().getProperty(PROP_API_KEY) || '';
  return raw.split(/[\r\n,;]+/)
            .map(function(k) { return k.trim(); })
            .filter(function(k) { return k.length > 0; });
}

/**
 * Obtém a primeira chave de API do Gemini (compatibilidade).
 */
function getGeminiApiKey() {
  const keys = getGeminiApiKeys();
  return keys.length > 0 ? keys[0] : '';
}

/**
 * Salva uma ou mais chaves de API do Gemini (uma por linha) de forma segura.
 */
function salvarGeminiApiKey(key) {
  exigirAdmin(); // Configuração de chaves restrita a administradores
  const props = PropertiesService.getScriptProperties();
  const keys = String(key || '').split(/[\r\n,;]+/)
                 .map(function(k) { return k.trim(); })
                 .filter(function(k) { return k.length > 0; });

  if (keys.length > 0) {
    props.setProperty(PROP_API_KEY, keys.join('\n'));
    const plural = keys.length > 1;
    return {
      success: true,
      message: plural
        ? (keys.length + " chaves de API cadastradas. O sistema fará rotação automática ao atingir o limite de cada uma.")
        : "Chave API do Gemini atualizada com sucesso no ambiente corporativo Cresol!"
    };
  } else {
    props.deleteProperty(PROP_API_KEY);
    return { success: true, message: "Chaves API removidas. O sistema passará a operar em modo de simulação integrada." };
  }
}

/**
 * Obtém as configurações do sistema para integração com Planilhas Google
 */
function getSystemConfig() {
  const keysCount = getGeminiApiKeys().length;
  const defaultConf = {
    spreadsheetUrl: '',
    abaPrincipal: 'DB_IMPORTACOES',
    abaAlertas: 'RISCO_LOG',
    driveFolderId: '',
    geminiKeyConfigured: keysCount > 0,
    geminiKeysCount: keysCount
  };

  const raw = PropertiesService.getScriptProperties().getProperty(PROP_CONFIG);
  if (raw) {
    try {
      const parsed = JSON.parse(raw);
      parsed.geminiKeyConfigured = keysCount > 0;
      parsed.geminiKeysCount = keysCount;
      return parsed;
    } catch (e) {
      return defaultConf;
    }
  }
  return defaultConf;
}

/**
 * Salva as configurações de integração do Google Sheets
 */
function salvarSystemConfig(config) {
  exigirAdmin(); // Configurações restritas a administradores
  const systemConfig = {
    spreadsheetUrl: config.spreadsheetUrl || '',
    abaPrincipal: config.abaPrincipal || 'DB_IMPORTACOES',
    abaAlertas: config.abaAlertas || 'RISCO_LOG',
    driveFolderId: config.driveFolderId || ''
  };
  
  PropertiesService.getScriptProperties().setProperty(PROP_CONFIG, JSON.stringify(systemConfig));
  
  // Inicializa a planilha se houver uma URL configurada
  if (systemConfig.spreadsheetUrl) {
    try {
      inicializarPlanilha(systemConfig.spreadsheetUrl, systemConfig.abaPrincipal, systemConfig.abaAlertas);
    } catch (e) {
      return { 
        success: true, 
        message: "Configuração salva, mas falhou ao inicializar cabeçalhos da planilha: " + e.toString(), 
        config: systemConfig 
      };
    }
  }
  
  return { success: true, message: "Configurações de mapeamento salvas com sucesso!", config: systemConfig };
}

/**
 * Inicializa os cabeçalhos das planilhas de destino caso não existam
 */
function inicializarPlanilha(url, abaPrincipalName, abaAlertasName) {
  const ss = SpreadsheetApp.openByUrl(url);
  
  // 1. Inicializa Aba Principal
  let sheetPrincipal = ss.getSheetByName(abaPrincipalName);
  if (!sheetPrincipal) {
    sheetPrincipal = ss.insertSheet(abaPrincipalName);
  }
  if (sheetPrincipal.getLastRow() === 0) {
    sheetPrincipal.appendRow([
      'Lote ID', 'Data e Hora', 'Arquivo', 'Comarca', 'Nº Matrícula',
      'Área', 'Endereço', 'Valor Estimado', 'Risco Geral', 'Rating',
      'Proprietários', 'Status de Liberação', 'Parecer Consolidado', 'Dados Completos (JSON)', 'Custo Total (R$)'
    ]);
    // Formatação do Cabeçalho (Verde Cresol institucional)
    const headerRange = sheetPrincipal.getRange(1, 1, 1, 15);
    headerRange.setBackground('#005c46')
               .setFontColor('#ffffff')
               .setFontWeight('bold')
               .setHorizontalAlignment('center');
    sheetPrincipal.setFrozenRows(1);
  }
  
  // 2. Inicializa Aba de Alertas
  let sheetAlertas = ss.getSheetByName(abaAlertasName);
  if (!sheetAlertas) {
    sheetAlertas = ss.insertSheet(abaAlertasName);
  }
  if (sheetAlertas.getLastRow() === 0) {
    sheetAlertas.appendRow([
      'Lote ID', 'Data e Hora', 'Nº Matrícula', 'Comarca', 'Título do Alerta', 'Descrição do Risco', 'Gravidade'
    ]);
    const headerRange = sheetAlertas.getRange(1, 1, 1, 7);
    headerRange.setBackground('#f58220') // Laranja Cresol
               .setFontColor('#ffffff')
               .setFontWeight('bold')
               .setHorizontalAlignment('center');
    sheetAlertas.setFrozenRows(1);
  }
}

/**
 * Persiste o histórico local com segurança.
 * O Apps Script limita cada propriedade a ~9KB, então mantemos apenas as análises mais
 * recentes e reduzimos progressivamente caso o limite ainda seja excedido.
 * A planilha do Google (quando configurada) permanece como o repositório durável e completo.
 */
function salvarBancoLocal(list) {
  const props = PropertiesService.getScriptProperties();
  let trimmed = list.slice(0, 15);
  try {
    props.setProperty('CRESOL_DB_ANALYSES', JSON.stringify(trimmed));
    return;
  } catch (e) {
    // Excedeu o limite de tamanho: reduz o número de registros até caber.
    while (trimmed.length > 1) {
      trimmed = trimmed.slice(0, trimmed.length - 1);
      try {
        props.setProperty('CRESOL_DB_ANALYSES', JSON.stringify(trimmed));
        return;
      } catch (e2) {
        // continua reduzindo
      }
    }
  }
}

/**
 * Lê o banco local (ScriptProperties) diretamente, SEM consultar a planilha.
 * É a fonte usada para localizar análises recém-criadas na hora de exportar/salvar,
 * já que a planilha pode ainda não conter o registro.
 */
function getBancoLocal() {
  const raw = PropertiesService.getScriptProperties().getProperty('CRESOL_DB_ANALYSES');
  if (raw) {
    try {
      const parsed = JSON.parse(raw);
      if (Array.isArray(parsed)) return parsed;
    } catch (e) { /* ignora */ }
  }
  return [];
}

/**
 * Localiza uma análise por id procurando primeiro no banco local (registros recentes)
 * e, se não achar, no histórico completo (planilha, quando configurada).
 */
function encontrarAnalise(id) {
  const locais = getBancoLocal();
  for (let i = 0; i < locais.length; i++) {
    if (locais[i] && locais[i].id === id) return locais[i];
  }
  const todas = listarAnalises();
  for (let j = 0; j < todas.length; j++) {
    if (todas[j] && todas[j].id === id) return todas[j];
  }
  return null;
}

/**
 * Carrega a lista histórica de análises.
 * Caso haja Planilha conectada, lê as análises dela. Caso contrário, retorna um histórico mockado robusto.
 */
function listarAnalises() {
  const config = getSystemConfig();
  
  if (config.spreadsheetUrl) {
    try {
      const ss = SpreadsheetApp.openByUrl(config.spreadsheetUrl);
      const sheet = ss.getSheetByName(config.abaPrincipal);
      if (sheet) {
        const rows = sheet.getDataRange().getValues();
        if (rows.length > 1) {
          const list = [];
          for (let i = rows.length - 1; i >= 1; i--) {
            const row = rows[i];

            // Preferência: reidrata o objeto completo salvo na coluna "Dados Completos (JSON)"
            if (row[13]) {
              try {
                const full = JSON.parse(row[13]);
                if (full && full.id) {
                  list.push(full);
                  continue;
                }
              } catch (e) {
                // JSON inválido: cai na reconstrução parcial abaixo
              }
            }

            // Fallback (linhas antigas sem a coluna de JSON completo)
            let props = [];
            try {
              props = JSON.parse(row[10]);
            } catch (e) {
              props = [{ nome: row[10], documento: '', participacao: '100%', status: 'REGULAR' }];
            }

            list.push({
              id: row[0],
              timestamp: row[1],
              nomeArquivo: row[2],
              status: row[11],
              metodo: 'Importado da Planilha',
              imovel: {
                matricula: row[4],
                comarca: row[3],
                valor: row[7],
                risco: row[8],
                rating: row[9],
                endereco: row[6],
                area: row[5],
                descricaoCurta: 'Recuperado do histórico oficial de planilhas.'
              },
              proprietarios: props,
              gravames: [],
              alertas: [], // Detalhes de alertas podem ser lidos da segunda aba se necessário
              cadeiaDominial: [],
              parecerConsolidado: row[12]
            });
          }
          return list;
        }
      }
    } catch (e) {
      // Ignora erro e cai no banco local
    }
  }

  // Fallback: banco local (ScriptProperties). Sem dados fictícios — começa vazio.
  return getBancoLocal();
}

/**
 * Salva as edições manuais de uma análise e marca o registro como VALIDADO.
 * Atualiza o banco local e, se houver planilha configurada, sincroniza a linha.
 */
function salvarAnalise(analysis) {
  if (!analysis || !analysis.id) {
    throw new Error("Análise inválida para salvamento.");
  }

  analysis.status = 'VALIDADO';
  analysis.validadoEm = new Date().toISOString();

  // Atualiza (ou insere) no banco local
  const list = getBancoLocal();
  let found = false;
  for (let i = 0; i < list.length; i++) {
    if (list[i] && list[i].id === analysis.id) {
      list[i] = analysis;
      found = true;
      break;
    }
  }
  if (!found) {
    list.unshift(analysis);
  }
  salvarBancoLocal(list);

  // Sincroniza com a planilha (upsert por id) quando configurada
  const config = getSystemConfig();
  if (config.spreadsheetUrl) {
    try {
      exportarParaSheets(analysis.id);
    } catch (e) {
      // Mantém o salvamento local mesmo se a planilha falhar
    }
  }

  return analysis;
}

/**
 * Exclui uma análise do histórico local
 */
function deletarAnalise(id) {
  const list = listarAnalises();
  const updated = list.filter(function(item) {
    return item.id !== id;
  });
  salvarBancoLocal(updated);
  
  // Também tenta remover da planilha se estiver configurado
  const config = getSystemConfig();
  if (config.spreadsheetUrl) {
    try {
      const ss = SpreadsheetApp.openByUrl(config.spreadsheetUrl);
      const sheet = ss.getSheetByName(config.abaPrincipal);
      if (sheet) {
        const rows = sheet.getDataRange().getValues();
        for (let i = 1; i < rows.length; i++) {
          if (rows[i][0] === id) {
            sheet.deleteRow(i + 1);
            break;
          }
        }
      }
    } catch (e) {
      // Erro silencioso na planilha
    }
  }
  
  return { success: true, message: "Registro excluído com sucesso!" };
}

/**
 * Exporta dados de uma análise específica para a Planilha do Google configurada
 */
function exportarParaSheets(analysisId) {
  const config = getSystemConfig();
  if (!config.spreadsheetUrl) {
    throw new Error("URL da planilha de destino não configurada nas configurações de mapeamento Cresol.");
  }
  
  // Busca a análise: primeiro no banco local (onde o registro recém-criado sempre existe),
  // depois no histórico da planilha. Evita "Análise não encontrada" logo após criar.
  const target = encontrarAnalise(analysisId);

  if (!target) {
    throw new Error("Análise não encontrada nos registros.");
  }
  
  try {
    const ss = SpreadsheetApp.openByUrl(config.spreadsheetUrl);
    
    // Assegura cabeçalhos
    inicializarPlanilha(config.spreadsheetUrl, config.abaPrincipal, config.abaAlertas);
    
    const sheetPrincipal = ss.getSheetByName(config.abaPrincipal);
    const sheetAlertas = ss.getSheetByName(config.abaAlertas);
    
    // Verifica se já existe para evitar duplicados
    const rows = sheetPrincipal.getDataRange().getValues();
    let rowIndexToUpdate = -1;
    for (let i = 1; i < rows.length; i++) {
      if (rows[i][0] === analysisId) {
        rowIndexToUpdate = i + 1;
        break;
      }
    }
    
    const rowData = [
      target.id,
      target.timestamp,
      target.nomeArquivo,
      target.imovel.comarca,
      target.imovel.matricula,
      target.imovel.area,
      target.imovel.endereco,
      target.imovel.valor,
      target.imovel.risco,
      target.imovel.rating,
      JSON.stringify(target.proprietarios),
      target.status,
      target.parecerConsolidado,
      JSON.stringify(target), // Dados completos para round-trip sem perda (confrontações, gravames, descrição legal, etc.)
      (typeof target.custoTotalBRL === 'number' ? Math.round(target.custoTotalBRL * 10000) / 10000 : '') // Custo Total (R$)
    ];

    if (rowIndexToUpdate > -1) {
      // Atualiza linha existente
      sheetPrincipal.getRange(rowIndexToUpdate, 1, 1, rowData.length).setValues([rowData]);
    } else {
      // Adiciona nova linha
      sheetPrincipal.appendRow(rowData);
    }
    
    // Exporta Alertas detalhados na aba de Alertas de Risco
    if (sheetAlertas && target.alertas && target.alertas.length > 0) {
      // Limpa alertas anteriores deste Lote para atualizar
      const alertaRows = sheetAlertas.getDataRange().getValues();
      for (let j = alertaRows.length - 1; j >= 1; j--) {
        if (alertaRows[j][0] === analysisId) {
          sheetAlertas.deleteRow(j + 1);
        }
      }
      
      // Insere novos alertas
      for (let k = 0; k < target.alertas.length; k++) {
        const alerta = target.alertas[k];
        sheetAlertas.appendRow([
          target.id,
          target.timestamp,
          target.imovel.matricula,
          target.imovel.comarca,
          alerta.titulo,
          alerta.descricao,
          alerta.tipo
        ]);
      }
    }
    
    return {
      success: true,
      message: "Dados sincronizados com sucesso na Planilha Google da Cresol!"
    };
    
  } catch (e) {
    throw new Error("Erro de conexão com o Google Sheets: " + e.toString());
  }
}

/**
 * Normaliza um texto para comparação: remove acentos (á->a, í->i, ç->c),
 * passa para minúsculas e mantém só alfanumérico.
 */
function normalizarChave(s) {
  return String(s == null ? '' : s)
    .normalize('NFD').replace(/[\u0300-\u036f]/g, '') // remove diacríticos/acentos
    .toLowerCase().replace(/[^a-z0-9]/g, '');
}

/**
 * Verifica se duas análises se referem à MESMA matrícula.
 * Regra: mesmo número de matrícula E mesma COMARCA (jurisdição do registro).
 * O texto do cartório varia muito entre leituras da IA, então não é usado como
 * chave — a comarca é o disciminador confiável. Matrículas de número igual em
 * comarcas diferentes são tratadas como imóveis DISTINTOS.
 */
function mesmaMatricula(a, b) {
  const ia = a.imovel || {}, ib = b.imovel || {};
  const mA = normalizarChave(ia.matricula), mB = normalizarChave(ib.matricula);
  if (!mA || !mB || mA !== mB) return false;
  const comA = normalizarChave(ia.comarca), comB = normalizarChave(ib.comarca);
  if (comA && comB) return comA === comB;
  // Sem comarca em algum dos lados: cai para o cartório como último recurso
  const cartA = normalizarChave(ia.cartorio), cartB = normalizarChave(ib.cartorio);
  return !!(cartA && cartB && cartA === cartB);
}

/**
 * Considera um valor "vazio" (placeholders de não identificado inclusos).
 */
function valorVazio(v) {
  if (v == null || v === '') return true;
  const s = String(v).toLowerCase();
  return s.indexOf('não identificad') > -1 || s === 'não avaliado' || s === 'não mensurada' || s === 'n/d' || s === 'não localizado';
}

/**
 * Mantém apenas os proprietários ATUAIS, descartando quem a IA por ventura marcou
 * como anterior/histórico (o dono anterior fica só na cadeia dominial).
 */
function apenasProprietariosAtuais(lista) {
  if (!Array.isArray(lista)) return [];
  return lista.filter(function(p) {
    if (!p) return false;
    const st = String(p.status || '').toLowerCase();
    if (st.indexOf('históric') > -1 || st.indexOf('historic') > -1 || st.indexOf('anterior') > -1) return false;
    return true;
  });
}

/**
 * Extrai a soma de áreas (em m²) mencionadas em um texto, convertendo hectares para m².
 * Reconhece formato brasileiro (43.425,25 m²) e unidades m²/m2/ha/hectare(s).
 */
function extrairAreaM2(texto) {
  if (!texto) return 0;
  const re = /([\d.]+(?:,\d+)?)\s*(m²|m2|metros?\s*quadrados?|hectares?|ha)/gi;
  let total = 0, m;
  while ((m = re.exec(String(texto))) !== null) {
    let num = parseFloat(m[1].replace(/\./g, '').replace(',', '.'));
    if (isNaN(num)) continue;
    const unidade = m[2].toLowerCase();
    if (unidade === 'ha' || unidade.indexOf('hectare') > -1) num = num * 10000; // hectare -> m²
    total += num;
  }
  return total;
}

/**
 * Formata uma área em m² no padrão brasileiro (ex.: 178.304,90 m²).
 */
function formatarAreaM2(m2) {
  const val = Math.round(m2 * 100) / 100;
  const inteiro = Math.floor(val);
  const dec = Math.round((val - inteiro) * 100);
  const decStr = dec < 10 ? ('0' + dec) : ('' + dec);
  const intStr = String(inteiro).replace(/\B(?=(\d{3})+(?!\d))/g, '.');
  return intStr + ',' + decStr + ' m²';
}

/**
 * Reforço: se a área total do imóvel não veio, tenta deduzi-la somando as áreas das
 * frações ideais dos proprietários (ex.: coproprietários com áreas delimitadas).
 * Só preenche quando há áreas explícitas por proprietário — evita cálculos incorretos.
 */
function preencherAreaTotal(analysis) {
  const imo = analysis.imovel || {};
  if (!valorVazio(imo.area)) return analysis; // já tem área total

  let somaM2 = 0;
  (analysis.proprietarios || []).forEach(function(p) {
    somaM2 += extrairAreaM2(p && p.participacao);
  });

  if (somaM2 > 0) {
    imo.area = formatarAreaM2(somaM2) + ' (somado das frações)';
    analysis.imovel = imo;
  }
  return analysis;
}

// Chaves de identificação de itens das listas (para comparar versões sem duplicar)
function chaveProprietario(p) { return normalizarChave(p.documento) || normalizarChave(p.nome); }
function chaveGravame(g) { return normalizarChave(g.registro) + '|' + normalizarChave(g.tipo); }
function chaveAlerta(a) { return normalizarChave(a.titulo); }
function chaveCadeia(c) { return normalizarChave(c.registro) || (normalizarChave(c.data) + normalizarChave(c.tipoTransacao)); }

/**
 * Une itens novos a uma lista existente, sem duplicar (por chave).
 */
function unirListas(destino, origem, chaveFn) {
  destino = Array.isArray(destino) ? destino : [];
  (origem || []).forEach(function(item) {
    const k = chaveFn(item);
    if (k && !destino.some(function(x) { return chaveFn(x) === k; })) destino.push(item);
  });
  return destino;
}

/**
 * Compara a versão nova de uma matrícula com a versão já existente e retorna a lista
 * de mudanças detectadas. Como as matrículas são cumulativas (registros e averbações
 * só são acrescentados), reportamos ADIÇÕES e ALTERAÇÕES de valores — não remoções,
 * evitando falsos positivos por falha de OCR.
 */
function compararAnalises(antiga, nova) {
  const mud = [];
  const ia = antiga.imovel || {}, ib = nova.imovel || {};
  const campos = [
    ['valor', 'Valor estimado'], ['risco', 'Risco'], ['rating', 'Rating'], ['nivelRisco', 'Nível de risco'],
    ['area', 'Área total'], ['areaConstruida', 'Área construída'], ['endereco', 'Endereço'],
    ['tipo', 'Tipo do imóvel'], ['cartorio', 'Cartório'], ['comarca', 'Comarca'],
    ['dataEmissaoCertidao', 'Data de emissão da certidão'], ['cadastroIncra', 'Cadastro INCRA/CCIR']
  ];
  campos.forEach(function(c) {
    const antes = ia[c[0]], depois = ib[c[0]];
    if (valorVazio(depois)) return; // a nova versão não informou -> ignora
    if (valorVazio(antes)) {
      mud.push({ categoria: 'Imóvel', descricao: c[1] + ' informado: "' + depois + '"' });
    } else if (String(antes).trim() !== String(depois).trim()) {
      mud.push({ categoria: 'Imóvel', descricao: c[1] + ' alterado: de "' + antes + '" para "' + depois + '"' });
    }
  });

  const propsAntigos = antiga.proprietarios || [];
  (nova.proprietarios || []).forEach(function(p) {
    const k = chaveProprietario(p);
    if (k && !propsAntigos.some(function(x) { return chaveProprietario(x) === k; })) {
      mud.push({ categoria: 'Titularidade', descricao: 'Novo proprietário: ' + (p.nome || '?') + (p.documento ? ' (' + p.documento + ')' : '') });
    }
  });

  const gravAntigos = antiga.gravames || [];
  (nova.gravames || []).forEach(function(g) {
    const k = chaveGravame(g);
    if (k && !gravAntigos.some(function(x) { return chaveGravame(x) === k; })) {
      mud.push({ categoria: 'Gravame', descricao: 'Novo gravame ' + (g.registro || '') + ': ' + (g.tipo || '') + (g.credor ? ' — ' + g.credor : '') });
    }
  });

  const cadAntiga = antiga.cadeiaDominial || [];
  (nova.cadeiaDominial || []).forEach(function(c) {
    const k = chaveCadeia(c);
    if (k && !cadAntiga.some(function(x) { return chaveCadeia(x) === k; })) {
      mud.push({ categoria: 'Cadeia dominial', descricao: 'Novo registro ' + (c.registro || '') + ': ' + (c.tipoTransacao || '') + (c.data ? ' (' + c.data + ')' : '') });
    }
  });

  const alAntigos = antiga.alertas || [];
  (nova.alertas || []).forEach(function(a) {
    const k = chaveAlerta(a);
    if (k && !alAntigos.some(function(x) { return chaveAlerta(x) === k; })) {
      mud.push({ categoria: 'Alerta', descricao: 'Novo alerta: ' + (a.titulo || '') });
    }
  });

  return mud;
}

/**
 * Reimportação da MESMA matrícula: detecta as mudanças, atualiza o registro para a
 * versão mais recente (mantendo dados já capturados) e acrescenta uma entrada datada
 * no histórico de alterações. Se houver mudança relevante, marca o status como ATUALIZADO.
 */
function atualizarComHistorico(existente, novo) {
  const mudancas = compararAnalises(existente, novo);

  // Campos escalares do imóvel -> versão mais recente (mantém o antigo se o novo vier vazio)
  const imoE = existente.imovel || {}, imoN = novo.imovel || {};
  Object.keys(imoN).forEach(function(k) {
    if (k === 'confrontacoes') return;
    if (!valorVazio(imoN[k])) imoE[k] = imoN[k];
  });
  imoE.confrontacoes = imoE.confrontacoes || [];
  (imoN.confrontacoes || []).forEach(function(c) {
    if (!imoE.confrontacoes.some(function(x) { return normalizarChave(x.posicao) === normalizarChave(c.posicao); })) {
      imoE.confrontacoes.push(c);
    }
  });
  existente.imovel = imoE;

  // Listas: união (nunca perde dados já capturados)
  existente.proprietarios = unirListas(existente.proprietarios, novo.proprietarios, chaveProprietario);
  existente.gravames = unirListas(existente.gravames, novo.gravames, chaveGravame);
  existente.alertas = unirListas(existente.alertas, novo.alertas, chaveAlerta);
  existente.cadeiaDominial = unirListas(existente.cadeiaDominial, novo.cadeiaDominial, chaveCadeia);

  if (!valorVazio(novo.parecerConsolidado)) existente.parecerConsolidado = novo.parecerConsolidado;
  if (novo.nomeArquivo) existente.nomeArquivo = novo.nomeArquivo;
  if (novo.metodo) existente.metodo = novo.metodo;

  preencherAreaTotal(existente);

  // Custo: acumula o gasto desta reimportação ao total da matrícula
  const custoNovoBRL = (novo.custo && novo.custo.brl) || 0;
  existente.custoTotalBRL = (existente.custoTotalBRL || 0) + custoNovoBRL;
  if (novo.custo) existente.custo = novo.custo; // custo do último processamento

  // Histórico datado (mais recente primeiro)
  existente.historico = Array.isArray(existente.historico) ? existente.historico : [];
  const agoraISO = new Date().toISOString();
  existente.historico.unshift({
    data: agoraISO,
    dataCertidao: (novo.imovel && novo.imovel.dataEmissaoCertidao) || '',
    arquivo: novo.nomeArquivo || '',
    mudancas: mudancas,
    custo: novo.custo || null
  });
  existente.atualizadoEm = agoraISO;

  // Marca como ATUALIZADO quando houve mudança relevante
  if (mudancas.length > 0) {
    existente.status = 'ATUALIZADO';
  }

  return existente;
}

/**
 * Registra uma nova análise OU, se a matrícula já existir, apenas acrescenta as
 * informações novas ao registro existente. Retorna o objeto efetivamente salvo.
 */
function registrarOuMesclar(finalResult) {
  preencherAreaTotal(finalResult); // deduz a área total se não veio da IA
  const list = listarAnalises();
  for (let i = 0; i < list.length; i++) {
    if (list[i] && mesmaMatricula(list[i], finalResult)) {
      // Reimportação: detecta mudanças, atualiza e registra no histórico
      const atualizado = atualizarComHistorico(list[i], finalResult);
      list[i] = atualizado;
      salvarBancoLocal(list);
      autoExportar(atualizado.id);
      return atualizado;
    }
  }
  // Primeira importação desta matrícula
  const custoInicial = (finalResult.custo && finalResult.custo.brl) || 0;
  finalResult.custoTotalBRL = custoInicial;
  finalResult.historico = [{
    data: new Date().toISOString(),
    dataCertidao: (finalResult.imovel && finalResult.imovel.dataEmissaoCertidao) || '',
    arquivo: finalResult.nomeArquivo || '',
    mudancas: [{ categoria: 'Cadastro', descricao: 'Primeira importação da matrícula.' }],
    custo: finalResult.custo || null
  }];
  list.unshift(finalResult);
  salvarBancoLocal(list);
  autoExportar(finalResult.id);
  return finalResult;
}

/**
 * Consulta o CEP na internet (ViaCEP) a partir do endereço e da comarca.
 * Usa a busca por UF/cidade/logradouro. Retorna o CEP ou '' se não localizar
 * (comum em imóveis rurais, que muitas vezes usam o CEP geral do município).
 */
function buscarCepPorEndereco(endereco, comarca) {
  try {
    if (!endereco || !comarca) return '';
    // Comarca no formato "Cidade - UF"
    const m = String(comarca).match(/^(.*?)[\s]*[-\/][\s]*([A-Za-z]{2})\s*$/);
    if (!m) return '';
    const cidade = m[1].trim();
    const uf = m[2].trim().toUpperCase();

    // Logradouro: trecho antes da primeira vírgula, sem números
    let logradouro = String(endereco).split(',')[0].trim();
    logradouro = logradouro.replace(/\d+/g, '').trim();
    if (logradouro.length < 3) return '';

    const url = 'https://viacep.com.br/ws/' + encodeURIComponent(uf) + '/' +
                encodeURIComponent(cidade) + '/' + encodeURIComponent(logradouro) + '/json/';
    const resp = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
    if (resp.getResponseCode() !== 200) return '';
    const data = JSON.parse(resp.getContentText());
    if (Array.isArray(data) && data.length > 0 && data[0].cep) {
      return data[0].cep;
    }
    return '';
  } catch (e) {
    return '';
  }
}

/**
 * Valida um CEP (geral do município) consultando o ViaCEP por CEP e conferindo
 * se a cidade/UF batem com a comarca. Retorna o CEP normalizado ou '' se não bater.
 */
function validarCepMunicipio(cep, comarca) {
  try {
    if (!cep || !comarca) return '';
    const soDigitos = String(cep).replace(/\D/g, '');
    if (soDigitos.length !== 8) return '';
    const m = String(comarca).match(/^(.*?)[\s]*[-\/][\s]*([A-Za-z]{2})\s*$/);
    if (!m) return '';
    const cidade = normalizarChave(m[1]);
    const uf = m[2].trim().toUpperCase();

    const resp = UrlFetchApp.fetch('https://viacep.com.br/ws/' + soDigitos + '/json/', { muteHttpExceptions: true });
    if (resp.getResponseCode() !== 200) return '';
    const data = JSON.parse(resp.getContentText());
    if (data && !data.erro && String(data.uf).toUpperCase() === uf && normalizarChave(data.localidade) === cidade) {
      return data.cep; // formato 00000-000
    }
    return '';
  } catch (e) {
    return '';
  }
}

/**
 * Cotação USD->BRL ao vivo (AwesomeAPI), com cache de 12h e fallback.
 */
function obterCambioUsdBrl() {
  const props = PropertiesService.getScriptProperties();
  const cacheRaw = props.getProperty('CRESOL_CAMBIO');
  if (cacheRaw) {
    try {
      const c = JSON.parse(cacheRaw);
      if (c && c.valor && c.em && (Date.now() - c.em) < 12 * 3600 * 1000) return c.valor;
    } catch (e) { /* ignora */ }
  }
  try {
    const resp = UrlFetchApp.fetch('https://economia.awesomeapi.com.br/last/USD-BRL', { muteHttpExceptions: true });
    if (resp.getResponseCode() === 200) {
      const data = JSON.parse(resp.getContentText());
      const bid = data && data.USDBRL && parseFloat(data.USDBRL.bid);
      if (bid && bid > 0) {
        props.setProperty('CRESOL_CAMBIO', JSON.stringify({ valor: bid, em: Date.now() }));
        return bid;
      }
    }
  } catch (e) { /* ignora e cai no fallback */ }
  if (cacheRaw) {
    try { const c = JSON.parse(cacheRaw); if (c && c.valor) return c.valor; } catch (e) {}
  }
  return CAMBIO_FALLBACK;
}

/**
 * Calcula o custo do processamento a partir do usageMetadata retornado pela API.
 * O custo de saída inclui os tokens de "raciocínio" (contabilizados no total).
 */
function calcularCusto(usage) {
  usage = usage || {};
  const inputTokens = usage.promptTokenCount || 0;
  const total = usage.totalTokenCount || 0;
  let outputTokens;
  if (total > inputTokens) {
    outputTokens = total - inputTokens;
  } else {
    outputTokens = (usage.candidatesTokenCount || 0) + (usage.thoughtsTokenCount || 0);
  }
  const usd = (inputTokens / 1e6) * PRECO_INPUT_USD_1M + (outputTokens / 1e6) * PRECO_OUTPUT_USD_1M;
  const cambio = obterCambioUsdBrl();
  return {
    inputTokens: inputTokens,
    outputTokens: outputTokens,
    totalTokens: inputTokens + outputTokens,
    usd: usd,
    brl: usd * cambio,
    cambio: cambio,
    modelo: GEMINI_MODEL_NAME,
    em: new Date().toISOString()
  };
}

/**
 * Persiste uma atualização de registro no banco local e re-sincroniza a planilha.
 */
function persistirAtualizacao(obj) {
  const list = getBancoLocal();

  for (let i = 0; i < list.length; i++) {
    if (list[i] && list[i].id === obj.id) { list[i] = obj; found = true; break; }

}

/**
 * Salva o arquivo importado (PDF/imagem) numa pasta do Drive, renomeado com o ID do Lote.
 * Retorna { id, url, nome } do arquivo criado.
 */
function salvarArquivoDrive(folderId, loteId, fileBase64, fileName, versao) {
  const folder = DriveApp.getFolderById(folderId);
  let mime = 'application/pdf';
  let b64 = fileBase64;
  if (fileBase64.indexOf('base64,') > -1) {
    const partes = fileBase64.split('base64,');
    const header = partes[0];
    b64 = partes[1];
    if (header.indexOf('image/png') > -1) mime = 'image/png';
    else if (header.indexOf('image/jpeg') > -1 || header.indexOf('image/jpg') > -1) mime = 'image/jpeg';
    else if (header.indexOf('application/pdf') > -1) mime = 'application/pdf';
  } else {
    const low = String(fileName || '').toLowerCase();
    if (low.endsWith('.png')) mime = 'image/png';
    else if (low.endsWith('.jpg') || low.endsWith('.jpeg')) mime = 'image/jpeg';
  }
  const ext = mime === 'image/png' ? 'png' : (mime === 'image/jpeg' ? 'jpg' : 'pdf');
  // Nome com o ID do Lote; em reimportações acrescenta a versão para não sobrescrever
  const nome = loteId + (versao && versao > 1 ? ('_v' + versao) : '') + '.' + ext;
  const bytes = Utilities.base64Decode(b64);
  const blob = Utilities.newBlob(bytes, mime, nome);
  const file = folder.createFile(blob);
  return { id: file.getId(), url: file.getUrl(), nome: nome };
}

/**
 * Extrai apenas a quantidade de área (texto) de uma fração ideal.
 * Ex.: "24,35% (43.425,25m²)" -> "43.425,25 m²". Se não houver área, retorna a fração original.
 */
function extrairAreaTexto(texto) {
  if (!texto) return '';
  const m = String(texto).match(/([\d.]+(?:,\d+)?)\s*(m²|m2|metros?\s*quadrados?|hectares?|ha)/i);
  if (!m) return String(texto).trim(); // sem área explícita: mantém o que veio (ex.: só "%")
  let unidade = m[2].toLowerCase();
  if (unidade === 'm2') unidade = 'm²';
  if (unidade === 'ha' || unidade.indexOf('hectare') > -1) unidade = 'ha';
  return m[1] + ' ' + unidade;
}

/**
 * Interpreta a data de emissão (texto livre da IA) e retorna um Date, ou null.
 * Aceita dd/mm/aaaa (com/sem hora), aaaa-mm-dd e "dd de <mês> de aaaa".
 */
function parseDataEmissao(texto) {
  if (!texto) return null;
  const s = String(texto).trim();
  let m = s.match(/(\d{1,2})\/(\d{1,2})\/(\d{4})/);
  if (m) return new Date(parseInt(m[3], 10), parseInt(m[2], 10) - 1, parseInt(m[1], 10));
  m = s.match(/(\d{4})-(\d{2})-(\d{2})/);
  if (m) return new Date(parseInt(m[1], 10), parseInt(m[2], 10) - 1, parseInt(m[3], 10));
  const meses = { 'janeiro': 0, 'fevereiro': 1, 'março': 2, 'marco': 2, 'abril': 3, 'maio': 4, 'junho': 5, 'julho': 6, 'agosto': 7, 'setembro': 8, 'outubro': 9, 'novembro': 10, 'dezembro': 11 };
  m = s.toLowerCase().match(/(\d{1,2})\s+de\s+([a-zç]+)\s+de\s+(\d{4})/);
  if (m && meses[m[2]] !== undefined) return new Date(parseInt(m[3], 10), meses[m[2]], parseInt(m[1], 10));
  return null;
}

/**
 * Formata um Date no padrão brasileiro dd/mm/aaaa.
 */
function formatarDataBR(d) {
  if (!(d instanceof Date) || isNaN(d.getTime())) return '';
  const dd = ('0' + d.getDate()).slice(-2);
  const mm = ('0' + (d.getMonth() + 1)).slice(-2);
  return dd + '/' + mm + '/' + d.getFullYear();
}

/**
 * Aptidão da certidão pela idade (emissão -> hoje):
 * até 30 dias -> "garantia"; até 1 ano -> "investimento"; até 5 anos -> "custeio";
 * acima de 5 anos (ou data ausente) -> vazio.
 */
function aptaPara(dataEmissao) {
  if (!(dataEmissao instanceof Date) || isNaN(dataEmissao.getTime())) return '';
  const hoje = new Date();
  if (dataEmissao > hoje) return '';
  const limGarantia = new Date(hoje.getTime()); limGarantia.setDate(limGarantia.getDate() - 30);
  const limInvest = new Date(hoje.getFullYear() - 1, hoje.getMonth(), hoje.getDate());
  const limCusteio = new Date(hoje.getFullYear() - 5, hoje.getMonth(), hoje.getDate());
  if (dataEmissao >= limGarantia) return 'garantia';
  if (dataEmissao >= limInvest) return 'investimento';
  if (dataEmissao >= limCusteio) return 'custeio';
  return 'Vencida';
}

/**
 * Agrupa os proprietários em unidades de titularidade: um CASAL (ambos "casado" e
 * com o mesmo sobrenome) vira uma única unidade; os demais ficam em unidades separadas.
 */
function agruparUnidades(props) {
  const ignorar = ['da', 'de', 'do', 'das', 'dos', 'e'];
  function sobrenome(nome) {
    const toks = String(nome || '').trim().split(/\s+/).filter(function(t) { return ignorar.indexOf(t.toLowerCase()) === -1; });
    return toks.length ? toks[toks.length - 1].toLowerCase() : '';
  }
  function ehCasado(p) {
    return String((p && p.estadoCivil) || '').toLowerCase().indexOf('casad') > -1;
  }

  const itens = (props || []).map(function(p) { return { p: p, usado: false }; });
  const unidades = [];
  for (let i = 0; i < itens.length; i++) {
    if (itens[i].usado) continue;
    itens[i].usado = true;
    const membros = [itens[i].p];
    if (ehCasado(itens[i].p)) {
      const sa = sobrenome(itens[i].p.nome);
      for (let j = i + 1; j < itens.length; j++) {
        if (itens[j].usado) continue;
        if (ehCasado(itens[j].p) && sa && sobrenome(itens[j].p.nome) === sa) {
          itens[j].usado = true;
          membros.push(itens[j].p);
          break; // um casal = 2 pessoas
        }
      }
    }
    unidades.push({ membros: membros });
  }
  return unidades;
}

/**
 * Monta as linhas do relatório (uma linha por matrícula), filtrando por período de importação.
 * Colunas: ID do Lote, Nº Matrícula, Proprietários (nomes juntos), Área Total (soma das partes),
 * Data Emissão Certidão (dd/mm/aaaa), Apta_para.
 */
function montarLinhasRelatorio(dataInicio, dataFim) {
  const inicio = dataInicio ? new Date(dataInicio + 'T00:00:00') : null;
  const fim = dataFim ? new Date(dataFim + 'T23:59:59') : null;

  const todas = listarAnalises();
  const linhas = [['id_lote', 'matricula', 'proprietarios', 'area_total', 'data_emissao_certidao', 'apta_para']];

  todas.forEach(function(a) {
    if (!a) return;
    let dt = null;
    try { dt = a.timestamp ? new Date(a.timestamp) : null; } catch (e) { dt = null; }
    if (inicio && dt && dt < inicio) return;
    if (fim && dt && dt > fim) return;
    if ((inicio || fim) && !dt) return; // sem data não entra em filtro por período

    const imo = a.imovel || {};
    const props = Array.isArray(a.proprietarios) ? a.proprietarios : [];
    const dataEmiss = parseDataEmissao(imo.dataEmissaoCertidao);
    const dataFmt = dataEmiss ? formatarDataBR(dataEmiss) : (imo.dataEmissaoCertidao || '');
    const apta = aptaPara(dataEmiss);

    // Agrupa por estado civil: casal (mesmo sobrenome) numa linha; demais separados
    const unidades = agruparUnidades(props);

    if (unidades.length === 0) {
      linhas.push([a.id || '', imo.matricula || '', '', imo.area || '', dataFmt, apta]);
    } else {
      unidades.forEach(function(u) {
        const nomes = u.membros.map(function(p) { return (p && p.nome) ? p.nome : ''; }).filter(String).join(' / ');

        // Área da unidade: soma das partes dos membros (para casal, soma as frações)
        const areasM2 = u.membros.map(function(p) { return extrairAreaM2(p && p.participacao); }).filter(function(v) { return v > 0; });
        let areaTexto;
        if (areasM2.length === 0) {
          // Sem área explícita nas frações: se há uma só unidade, é 100% do imóvel
          areaTexto = (unidades.length === 1) ? (imo.area || '') : u.membros.map(function(p) { return (p && p.participacao) || ''; }).filter(String).join(' + ');
        } else {
          const todosIguais = areasM2.every(function(v) { return v === areasM2[0]; });
          const somaM2 = todosIguais ? areasM2[0] : areasM2.reduce(function(s, v) { return s + v; }, 0);
          areaTexto = formatarAreaM2(somaM2);
        }

        linhas.push([a.id || '', imo.matricula || '', nomes, areaTexto, dataFmt, apta]);
      });
    }
  });

  return linhas;
}

/**
 * Gera o relatório em XLSX (Excel), uma linha por proprietário. Retorna { filename, mimeType, base64 }.
 */
function gerarRelatorioXLS(dataInicio, dataFim) {
  exigirAdmin(); // Relatório fica na aba Configurações (restrita a admin)
  const linhas = montarLinhasRelatorio(dataInicio, dataFim);
  const nCols = linhas[0].length;

  const ss = SpreadsheetApp.create('Relatorio_Cresol_tmp_' + new Date().getTime());
  const fileId = ss.getId();
  try {
    const sheet = ss.getSheets()[0];
    sheet.setName('Relatório');
    sheet.getRange(1, 1, linhas.length, nCols).setValues(linhas);
    const header = sheet.getRange(1, 1, 1, nCols);
    header.setBackground('#005c46').setFontColor('#ffffff').setFontWeight('bold');
    sheet.setFrozenRows(1);
    SpreadsheetApp.flush();

    const url = 'https://docs.google.com/spreadsheets/d/' + fileId + '/export?format=xlsx';
    const resp = UrlFetchApp.fetch(url, {
      headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() },
      muteHttpExceptions: true
    });
    const base64 = Utilities.base64Encode(resp.getBlob().getBytes());

    const hoje = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
    return {
      filename: 'Relatorio_Matriculas_' + hoje + '.xlsx',
      mimeType: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
      base64: base64,
      registros: linhas.length - 1
    };
  } finally {
    try { DriveApp.getFileById(fileId).setTrashed(true); } catch (e) {}
  }
}

/**
 * Gera o relatório em CSV (separador ';', UTF-8 com BOM para abrir corretamente no Excel).
 */
function gerarRelatorioCSV(dataInicio, dataFim) {
  exigirAdmin();
  const linhas = montarLinhasRelatorio(dataInicio, dataFim);
  const sep = ';';
  const csv = linhas.map(function(row) {
    return row.map(function(c) {
      let s = (c == null) ? '' : String(c);
      if (s.indexOf('"') > -1) s = s.replace(/"/g, '""');
      if (s.indexOf(sep) > -1 || s.indexOf('"') > -1 || s.indexOf('\n') > -1 || s.indexOf('\r') > -1) s = '"' + s + '"';
      return s;
    }).join(sep);
  }).join('\r\n');
  const comBom = String.fromCharCode(0xFEFF) + csv; // BOM UTF-8 para acentuação correta no Excel
  const base64 = Utilities.base64Encode(comBom, Utilities.Charset.UTF_8);

  const hoje = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd');
  return {
    filename: 'Relatorio_Matriculas_' + hoje + '.csv',
    mimeType: 'text/csv;charset=utf-8',
    base64: base64,
    registros: linhas.length - 1
  };
}

      "6. Identificadores: se a matrícula citar o CNM (Código Nacional de Matrícula, padrão do ONR/SREI, formato tipo NNNNNN.N.NNNNNNN-NN), transcreva-o no campo 'cnm'; se citar CEP do imóvel, transcreva-o no campo 'cep'. NÃO invente CNM nem CEP específico se não constarem. Adicionalmente, preencha 'cepMunicipio' com o CEP GERAL do município (a localidade/sede, conforme os Correios) — este servirá de referência quando o CEP específico do imóvel não constar.\n\n" +
      "LEITURA DO DOCUMENTO (OCR): O documento pode chegar como PDF nativo, PDF digitalizado ou FOTOGRAFIA da matrícula (torta, com sombras, baixa resolução, carimbos ou trechos manuscritos). Faça a transcrição visual (OCR) integral, lendo todos os R- e AV- na ordem. NUNCA invente dados: se um campo estiver ilegível ou ausente, use 'Não identificado' e registre a ilegibilidade como alerta do tipo ATENÇÃO. Baseie-se estritamente no documento fornecido.";

    let prompt = "Faça a análise técnica jurídica profunda desta matrícula imobiliária (PDF, imagem ou transcrição) aplicando a metodologia de validação cruzada. Retorne ESTRITAMENTE um objeto JSON válido no formato abaixo, sem texto fora do JSON.\n\n" +
                 "FORMATO JSON SOLICITADO:\n" +
                 "{\n" +
                 "  \"imovel\": {\n" +
                 "    \"matricula\": \"string\",\n" +
                 "    \"cartorio\": \"string (Ofício/Registro de Imóveis)\",\n" +
                 "    \"comarca\": \"string (Comarca - UF)\",\n" +

                 "    \"cadastroIncra\": \"string (INCRA/CCIR se rural, senão vazio)\",\n" +
                 "    \"valor\": \"string (Valor estimado de mercado)\",\n" +
                 "    \"risco\": \"string (Ex: 3.5/10)\",\n" +
                 "    \"rating\": \"string (AAA+ a D)\",\n" +
                 "    \"nivelRisco\": \"string (BAIXO|MÉDIO|ALTO conforme os ônus ativos)\",\n" +
                 "    \"endereco\": \"string\",\n" +
                 "    \"cep\": \"string (CEP do imóvel, SOMENTE se constar no documento; senão vazio)\",\n" +
                 "    \"cepMunicipio\": \"string (CEP geral do município/sede, conforme os Correios; referência para quando o CEP específico não constar)\",\n" +
                 "    \"cnm\": \"string (Código Nacional de Matrícula - CNM/ONR, SOMENTE se citado na matrícula; senão vazio)\",\n" +

                 "    \"descricaoCurta\": \"string (resumo objetivo do imóvel)\",\n" +
                 "    \"descricaoLegal\": \"string (descritivo legal completo e atualizado, pronto para contrato)\",\n" +

                 "      { \"posicao\": \"Norte|Sul|Leste|Oeste\", \"confrontante\": \"string\", \"medida\": \"string\" }\n" +

                 "    { \"nome\": \"string (SOMENTE proprietários ATUAIS)\", \"documento\": \"string (CPF/CNPJ ou 'Não informado')\", \"participacao\": \"string (% e área)\", \"descricaoArea\": \"string (descrição COMPLETA da área/fração deste proprietário: quantidade, localização/delimitação dentro do imóvel e registro de origem)\", \"estadoCivil\": \"string\", \"regimeCasamento\": \"string (comunhão universal|comunhão parcial|separação total|separação obrigatória|participação final nos aquestos|união estável|não se aplica)\", \"formaAquisicao\": \"string (compra e venda|doação|herança|partilha|adjudicação|usucapião|dação em pagamento|permuta|outra)\", \"validacaoTitularidade\": \"string (conclusão jurídica: comunicação/meação do cônjuge conforme o regime e a forma de aquisição, necessidade de outorga conjugal - art. 1.647 CC)\", \"status\": \"REGULAR|ATENÇÃO\" }\n" +

    // Garante a estrutura mínima de "imovel" para não quebrar a renderização na interface

        tipo: imovelSafe.tipo || "Não identificado",
        dataEmissaoCertidao: imovelSafe.dataEmissaoCertidao || "Não identificada",

        risco: imovelSafe.risco || "N/D",

        endereco: imovelSafe.endereco || "Não identificado",
        cep: imovelSafe.cep || "",
        cepMunicipio: imovelSafe.cepMunicipio || "",
        cnm: imovelSafe.cnm || "",
        area: imovelSafe.area || "Não identificado",
        areaConstruida: imovelSafe.areaConstruida || "Não mensurada",
        descricaoCurta: imovelSafe.descricaoCurta || "Descrição não extraída do documento.",
        descricaoLegal: imovelSafe.descricaoLegal || "",
        confrontacoes: imovelSafe.confrontacoes || []
      },
      proprietarios: apenasProprietariosAtuais(parsedData.proprietarios || []),
      gravames: parsedData.gravames || [],
      alertas: parsedData.alertas || [],
      cadeiaDominial: parsedData.cadeiaDominial || [],
      parecerConsolidado: parsedData.parecerConsolidado || "Análise concluída sem parecer textual.",
      status: parsedData.status || "EM REVISÃO"
    };

    // Custo real deste processamento (tokens reais reportados pela API + câmbio ao vivo)
    finalResult.custo = calcularCusto(parsedRaw.usageMetadata);

    // CEP: (1) do documento; (2) ViaCEP pelo logradouro; (3) CEP geral do município (validado)
    if (!finalResult.imovel.cep) {
      let cepFinal = buscarCepPorEndereco(finalResult.imovel.endereco, finalResult.imovel.comarca);
      let fonte = 'consultado';
      if (!cepFinal && finalResult.imovel.cepMunicipio) {
        const cepMun = validarCepMunicipio(finalResult.imovel.cepMunicipio, finalResult.imovel.comarca);
        if (cepMun) { cepFinal = cepMun; fonte = 'CEP geral do município'; }
      }
      finalResult.imovel.cep = cepFinal ? (cepFinal + ' (' + fonte + ')') : 'Não localizado';
    }

    // Grava (ou mescla, se a matrícula já existir) no banco
    const resultado = registrarOuMesclar(finalResult);

    // Salva o arquivo importado no Drive (renomeado com o ID do Lote), se configurado
    if (fileBase64 && fileBase64.trim() !== "") {
      try {
        const cfg = getSystemConfig();
        if (cfg.driveFolderId) {
          const versao = Array.isArray(resultado.historico) ? resultado.historico.length : 1;
          const info = salvarArquivoDrive(cfg.driveFolderId, resultado.id, fileBase64, name, versao);
          resultado.arquivoDrive = info;
          persistirAtualizacao(resultado); // grava o link do arquivo no registro
        }
      } catch (e) {
        // Não interrompe a análise se o Drive falhar (ex.: pasta inválida ou sem permissão)
      }
    }

    return resultado;

  } catch (err) {
    throw new Error("Falha ao se comunicar com o cérebro da Inteligência Artificial: " + err.toString());
  }
}
